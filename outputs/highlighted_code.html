30// --- Function definition with parameters and local vars ---
31int factorial(int n) {
32    if (n <= 1) {
33        return 1;
34    }
35    return n * factorial(n - 1);
36}

44// --- Function with array parameter and for loop ---
45int sum_array(int arr[], int size) {
46    int sum = 0;
47    for (int i = 0; i < size; i = i + 1) {
48        sum = sum + arr[i];
49    }
50    return sum;

53// --- Function with while loop and break/continue ---
54int count_even(int arr[], int size) {
55    int count = 0;
56    int i = 0;
57    while (i < size) {
58        if (arr[i] == 0) {
59            i = i + 1;
60            continue;   // skip zero
61        }
62        if (arr[i] % 2 == 0) {
63            count = count + 1;
64        }
65        i = i + 1;
66    }
67    return count;
68}

84int power(int base, int exp) {
85    if (exp == 0) {

87    }
